`timescale 1us / 1ps

module pwm(
  input wire CLK,
  input wire[3:0] BTN,
  input wire [5:0] SW,
  output wire [3:0] PWM_OUT,
  output wire TEST
    );
    
     reg [6:0] counter = 0;
     reg [7:0] up_down_counter =0;
     reg direction = 1; //1-up 0-down
     reg [3:0]PWM  = 0;
     
     reg [4:0] cycle1 =0;
     reg [1:0] mode1  =0; //0-left alligned 1-right allined 2-center
     reg inv1=0;
     
     reg [4:0] cycle2  =0;
     reg [1:0] mode2  =0;
     reg inv2=0;
     
     reg [4:0] cycle3  =0;
     reg [1:0] mode3  =0;
     reg inv3=0;
     
     reg [4:0] cycle4  =0;
     reg [1:0] mode4  =0;
     reg inv4=0;
     
     reg [3:0] debounce1=0;
     reg [3:0] debounce2=0;
     reg [3:0] debounce3=0;
     reg [3:0] debounce4=0;
     reg [3:0] MY_BTN=0;
     
     
     reg testowy=0;
     
 always @(posedge CLK) begin
 
    if(counter<100)             //counter incrementation 
        counter <= counter +1;
    else
        counter <=0;
        
    if(up_down_counter == 50)
        direction=0;
        
    if (up_down_counter == 0)
        direction=1;
        
    if(direction==1)
        up_down_counter <= up_down_counter+1;
    else
        up_down_counter <= up_down_counter-1;
                                                //counters end here
   //ch1                
   if (mode1==0) begin   //pwm ch1 left alligned
        if (cycle1*10>counter)
            PWM[0] = 1;
        else
            PWM[0]=0;
   
   end else if (mode1==1)begin //pwm ch1 right alligned
        if (cycle1*10>99-counter)
            PWM[0] = 1;
        else
            PWM[0]=0;
   end else begin //pwm ch1 center alligned
        if (50-cycle1*5 <up_down_counter)
            PWM[0]=1;
        else
            PWM[0]=0;
   end
   if (inv1==1) PWM[0]=~PWM[0];
   
   
   //ch2                
   if (mode2==0) begin   //pwm ch2 left alligned
        if (cycle2*10>counter)
            PWM[1] = 1;
        else
            PWM[1]=0;
   
   end else if (mode2==1)begin //pwm ch2 right alligned
        if (cycle2*10>99-counter)
            PWM[1] = 1;
        else
            PWM[1]=0;
   end else begin //pwm ch2 center alligned
        if (50-cycle2*5 <up_down_counter)
            PWM[1]=1;
        else
            PWM[1]=0;
   end
   if (inv2==1) PWM[1]=~PWM[1];
   
   //ch3                
   if (mode3==0) begin   //pwm ch3 left alligned
        if (cycle3*10>counter)
            PWM[2] = 1;
        else
            PWM[2]=0;
   
   end else if (mode3==1)begin //pwm ch3 right alligned
        if (cycle3*10>99-counter)
            PWM[2] = 1;
        else
            PWM[2]=0;
   end else begin //pwm ch3 center alligned
        if (50-cycle3*5 <up_down_counter)
            PWM[2]=1;
        else
            PWM[2]=0;
   end
   if (inv3==1) PWM[2]=~PWM[2];
   
   //ch4                
   if (mode4==0) begin   //pwm ch4 left alligned
        if (cycle4*10>counter)
            PWM[3] = 1;
        else
            PWM[3]=0;
   
   end else if (mode1==1)begin //pwm ch4 right alligned
        if (cycle4*10>99-counter)
            PWM[3] = 1;
        else
            PWM[3]=0;
   end else begin //pwm ch4 center alligned
        if (50-cycle4*5 <up_down_counter)
            PWM[3]=1;
        else
            PWM[3]=0;
   end
   if (inv4==1) PWM[3]=~PWM[3];
   
   
   //debounce
   debounce1 <= debounce1<<1+BTN[0];
   debounce2 <= debounce2<<1+BTN[1];
   debounce3 <= debounce3<<1+BTN[2];
   debounce4 <= debounce4<<1; debounce4[0]=BTN[3];
   
   if(debounce1==4'b1111) MY_BTN[0]=1; else MY_BTN[0]=0;
   if(debounce2==4'b1111) MY_BTN[1]=1; else MY_BTN[1]=0;
   if(debounce3==4'b1111) MY_BTN[2]=1; else MY_BTN[2]=0;
   if(debounce4==4'b1111) MY_BTN[3]=1; else MY_BTN[3]=0;
   
   
              
 
end


always @(posedge BTN[0]) begin
    if (SW==0) begin 
        if (cycle1==9)
            cycle1<=0;
        else
            cycle1 <= cycle1+1;
    end
    
    if (SW==1) begin 
        if (cycle2==9)
            cycle2<=0;
        else
            cycle2 <= cycle2+1;
    end
    
    if (SW==2) begin 
        if (cycle3==9)
            cycle3<=0;
        else
            cycle3 <= cycle3+1;
    end
    
    if (SW==3) begin 
        if (cycle4==9)
            cycle4<=0;
        else
            cycle4 <= cycle4+1;
    end
end

always @(posedge BTN[1]) begin
    if (SW==0) begin 
        if (mode1==2)
            mode1<=0;
        else
            mode1 <= mode1+1;
    end
    
    if (SW==1) begin 
        if (mode2==2)
            mode2<=0;
        else
            mode2 <= mode2+1;
    end
    
    if (SW==2) begin 
        if (mode3==2)
            mode3<=0;
        else
            mode3 <= mode3+1;
    end
    
    if (SW==3) begin 
        if (mode4==2)
            mode4<=0;
        else
            mode4 <= mode4+1;
    end
end

always @(posedge BTN[2]) begin
    if (SW==0) inv1=~inv1;
    
    if (SW==1) inv2=~inv2;
    
    if (SW==2) inv3=~inv3;
    
    if (SW==3) inv4=~inv4;
end

always @(posedge inv1) begin
end

always @(posedge BTN[3]) testowy=~testowy;


assign PWM_OUT[0] = PWM[0];
assign PWM_OUT[1] = PWM[1];
assign PWM_OUT[2] = PWM[2];
assign PWM_OUT[3] = PWM[3];
assign TEST = MY_BTN[3];


endmodule
